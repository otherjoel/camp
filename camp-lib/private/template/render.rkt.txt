#lang racket/base

(require camp
         punct/doc
         punct/fetch)

(provide render-page
         render-post)

(define (layout title body-content)
  `(html
    (head
     (meta ((charset "utf-8")))
     (meta ((name "viewport") (content "width=device-width, initial-scale=1")))
     (title ,title)
     (link ((rel "stylesheet") (href ,(format "/style.css?~a" (current-seconds)))))
     (link ((rel "alternate") (type "application/atom+xml") (href "/feed.atom"))))
    (body
     (div ((class "paper"))
      (header
       (nav (a ((href "/")) "Home") " · " (a ((href "/feed.atom")) "Feed")))
      (main ,@"@"body-content)
      (footer (p "Powered by Camp"))))))

(define (render-page doc ctxt)
  (define title (or (meta-ref doc 'title) "Untitled"))
  (define body (camp-doc->html-xexpr doc))
  (define slug (context-slug ctxt))

  ;; On the home page, show recent blog posts
  (define blog-listing
    (if (equal? slug "index")
        (let ([posts (get-collection "blog")])
          `((section ((class "posts"))
             (h2 "Recent Posts")
             (ul
              ,@"@"(for/list ([p posts])
                   `(li
                     (a ((href ,(page-link-url p)))
                        ,(page-link-title p))
                     ,(let ([date (hash-ref (page-link-metas p) 'date #f)])
                        (if date
                            `(span ((class "date")) " · " ,(~d "d MMM yyyy" date))
                            ""))))))))
        '()))

  (layout title `((article (h1 ,title) ,@"@"body ,@"@"blog-listing))))

(define (render-post doc ctxt)
  (define title (or (meta-ref doc 'title) "Untitled"))
  (define date (meta-ref doc 'date))
  (define body (camp-doc->html-xexpr doc))
  (layout title
          `((article ((class "post"))
             (h1 ,title)
             ,(if date `(time ((datetime ,(~d "yyyy-MM-dd" date))) ,(~d "d MMM yyyy" date)) "")
             ,@"@"body
             (nav ((class "pagination"))
                  ,(let ([p (prev ctxt)])
                     (if p `(a ((href ,(page-link-url p))) "← " ,(page-link-title p)) ""))
                  ,(let ([n (next ctxt)])
                     (if n `(a ((href ,(page-link-url n))) ,(page-link-title n) " →") "")))))))
